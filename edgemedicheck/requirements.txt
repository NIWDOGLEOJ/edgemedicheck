# EdgeMediCheck runtime dependencies

# Core pipeline -- required on both laptop and Raspberry Pi.
# Upper bound is deliberate. OpenCV 5.0 reordered the return tuple of
# BarcodeDetector.detectAndDecodeMulti (4.x: retval, info, type, points;
# 5.0: retval, info, points, straight_code), so an unbounded ">=4.8.0" quietly
# installed 5.0 and fed corner coordinates into the barcode symbology field.
# barcode.py now detects the layout at runtime, but pin the major version so a
# future release cannot change the API underneath a Raspberry Pi deployment.

numpy>=1.24.0
